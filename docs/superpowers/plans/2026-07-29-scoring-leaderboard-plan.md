# Multi-user Scoring & Leaderboard Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add Supabase-backed accounts, a richer score formula, and a server-validated global leaderboard to NILEFRONT, per `docs/superpowers/specs/2026-07-29-scoring-leaderboard-design.md`.

**Architecture:** `index.html` stays a single static file (no build step) and gains `supabase-js` via CDN for auth/session/leaderboard reads. A new Supabase Edge Function (`submit-score`) is the only thing allowed to write to the `runs` table — it re-derives the score from raw stats the client sends, so the leaderboard can't be forged by editing client JS or the network request.

**Tech Stack:** Vanilla JS + three.js (existing), `@supabase/supabase-js` v2 (CDN), Supabase Postgres + Auth + Edge Functions (Deno), Supabase CLI (`supabase`).

## Global Constraints

- No Docker on this machine — never use `supabase start` or any local dev stack. All schema/function work targets the real hosted Supabase project directly (link + push/deploy).
- The Supabase CLI is installed via Homebrew at `/opt/homebrew/bin/supabase` (v2.110.0) — invoke it directly as `supabase` (no `npx` needed).
- No build step for the game itself. `index.html` stays a single file; `supabase-js` loads via `<script>` CDN tag, same pattern as three.js.
- The client-side Supabase client instance must be named `sb`, not `supabase` — the UMD CDN build exposes a global `window.supabase` namespace (`supabase.createClient(...)`), and shadowing it breaks that call.
- Only the Edge Function's service-role client (key auto-provisioned by Supabase into the function's environment, never present in `index.html`) may write to `runs` or `profiles`. No client-side INSERT/UPDATE/DELETE policy is added for either table.
- Login is required before START; there is no guest/anonymous play path.
- A score submission fires on every `onAllDead()` call (regular wave, Nile Wave, or boss clear) — not on death, since death only respawns the player and `player.score` keeps accumulating.
- Before writing any Supabase-specific code (trigger SQL, RLS policies, Edge Function client construction, CDN URL), fetch the current docs page named in that task — API shapes and CDN paths move between Supabase versions, per the supabase skill's guidance not to rely on training data.

```sql
-- profiles: one row per auth user, public display name for the leaderboard
create table public.profiles (
  id uuid primary key references auth.users(id) on delete cascade,
  username text not null unique,
  created_at timestamptz not null default now()
);

alter table public.profiles enable row level security;

create policy "profiles are publicly readable"
on public.profiles for select
to anon, authenticated
using (true);
-- deliberately no insert/update/delete policy for anon/authenticated:
-- rows are only ever created by the trigger below

create or replace function public.handle_new_user()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
begin
  insert into public.profiles (id, username)
  values (new.id, new.raw_user_meta_data->>'username');
  return new;
end;
$$;

create trigger on_auth_user_created
  after insert on auth.users
  for each row execute function public.handle_new_user();

-- runs: one row per wave-clear checkpoint submitted by the submit-score Edge Function
create table public.runs (
  id bigint generated always as identity primary key,
  user_id uuid not null references public.profiles(id) on delete cascade,
  score int not null,
  wave_reached int not null,
  era_reached int not null,
  kills int not null,
  boss_kills int not null,
  shots_fired int not null,
  shots_hit int not null,
  max_streak int not null,
  duration_seconds numeric not null,
  created_at timestamptz not null default now()
);

alter table public.runs enable row level security;

create policy "runs are publicly readable"
on public.runs for select
to anon, authenticated
using (true);
-- deliberately no insert/update/delete policy for anon/authenticated:
-- only the service-role client inside the submit-score Edge Function may write here

create view public.leaderboard
with (security_invoker = true) as
select distinct on (r.user_id)
  p.username, r.score, r.wave_reached, r.era_reached, r.created_at
from public.runs r
join public.profiles p on p.id = r.user_id
order by r.user_id, r.score desc, r.created_at asc;
```

```typescript
import { createClient } from 'https://esm.sh/@supabase/supabase-js@2'

const SUPABASE_URL = Deno.env.get('SUPABASE_URL')!
const ANON_KEY = Deno.env.get('SUPABASE_ANON_KEY')!
const SERVICE_ROLE_KEY = Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!

interface RunStats {
  kills: number
  bossKills: number
  maxStreak: number
  wavesCleared: number
  bossesDefeated: number
  shotsFired: number
  shotsHit: number
  durationSeconds: number
}

// Same shape as the client's live-preview copy in index.html — the client's number is only
// a preview; this is the one that actually gets stored.
function computeScore(s: RunStats): number {
  const streakMult = 1 + Math.min(Math.floor(s.maxStreak / 5) * 0.1, 1)
  const killPoints = Math.round((s.kills * 1 + s.bossKills * 10) * streakMult)
  const waveBonus = s.wavesCleared * 20 + s.bossesDefeated * 150
  const accuracyBonus = s.shotsFired > 0 ? Math.round(200 * (s.shotsHit / s.shotsFired)) : 0
  const parSeconds = 45 + s.wavesCleared * 30
  const speedBonus =
    s.durationSeconds > 0 && s.durationSeconds < parSeconds
      ? Math.round(150 * (1 - s.durationSeconds / parSeconds))
      : 0
  return killPoints + waveBonus + accuracyBonus + speedBonus
}

function isNonNegInt(n: unknown): n is number {
  return typeof n === 'number' && Number.isInteger(n) && n >= 0
}

function jsonResponse(body: unknown, status: number): Response {
  return new Response(JSON.stringify(body), {
    status,
    headers: { 'Content-Type': 'application/json' },
  })
}

Deno.serve(async (req) => {
  if (req.method !== 'POST') return jsonResponse({ error: 'method not allowed' }, 405)

  const authHeader = req.headers.get('Authorization')
  if (!authHeader) return jsonResponse({ error: 'missing Authorization header' }, 401)

  const userClient = createClient(SUPABASE_URL, ANON_KEY, {
    global: { headers: { Authorization: authHeader } },
  })
  const {
    data: { user },
    error: authError,
  } = await userClient.auth.getUser()
  if (authError || !user) return jsonResponse({ error: 'invalid session' }, 401)

  let body: Record<string, unknown>
  try {
    body = await req.json()
  } catch {
    return jsonResponse({ error: 'invalid JSON body' }, 400)
  }

  const {
    kills, bossKills, maxStreak, wavesCleared, bossesDefeated,
    shotsFired, shotsHit, durationSeconds, waveReached, eraReached,
  } = body as Record<string, number>

  const valid =
    isNonNegInt(kills) &&
    isNonNegInt(bossKills) && bossKills <= 2 &&
    isNonNegInt(maxStreak) &&
    isNonNegInt(wavesCleared) && wavesCleared <= 6 &&
    isNonNegInt(bossesDefeated) && bossesDefeated <= 2 &&
    isNonNegInt(shotsFired) &&
    isNonNegInt(shotsHit) && shotsHit <= shotsFired &&
    typeof durationSeconds === 'number' && durationSeconds > 0 &&
    durationSeconds <= 4 * 3600 && durationSeconds >= wavesCleared * 2 &&
    isNonNegInt(waveReached) && waveReached >= 1 && waveReached <= 6 &&
    isNonNegInt(eraReached) && eraReached <= 1

  if (!valid) return jsonResponse({ error: 'stats failed validation' }, 400)

  const stats: RunStats = { kills, bossKills, maxStreak, wavesCleared, bossesDefeated, shotsFired, shotsHit, durationSeconds }
  const score = computeScore(stats)

  const admin = createClient(SUPABASE_URL, SERVICE_ROLE_KEY)
  const { error: insertError } = await admin.from('runs').insert({
    user_id: user.id,
    score,
    wave_reached: waveReached,
    era_reached: eraReached,
    kills,
    boss_kills: bossKills,
    shots_fired: shotsFired,
    shots_hit: shotsHit,
    max_streak: maxStreak,
    duration_seconds: durationSeconds,
  })
  if (insertError) return jsonResponse({ error: insertError.message }, 500)

  const { count } = await admin
    .from('leaderboard')
    .select('*', { count: 'exact', head: true })
    .gt('score', score)

  return jsonResponse({ score, rank: (count ?? 0) + 1 }, 200)
})
```

```javascript
function computeScore(s){
  const streakMult = 1 + Math.min(Math.floor(s.maxStreak/5)*0.1, 1);
  const killPoints = Math.round((s.kills*1 + s.bossKills*10) * streakMult);
  const waveBonus = s.wavesCleared*20 + s.bossesDefeated*150;
```

```javascript
async function submitScore(){
  const { data: { session } } = await sb.auth.getSession();
  if(!session){
    setBanner("Couldn't save to leaderboard — signed out"); setTimeout(()=>{ if(!gameWon) setBanner(''); },1800);
    return;
  }
  const stats = {
    kills: runStats.kills,
    bossKills: runStats.bossKills,
    maxStreak: runStats.maxStreak,
    wavesCleared: waveIdx + 1,
    bossesDefeated: bossNum,
    shotsFired: runStats.shotsFired,
    shotsHit: runStats.shotsHit,
    durationSeconds: performance.now()*0.001 - runStats.runStartTime,
    waveReached: waveIdx + 1,
    eraReached: currentEra,
  };
  try {
    const res = await fetch(SUPABASE_URL + '/functions/v1/submit-score', {
      method: 'POST',
      headers: { 'Content-Type':'application/json', 'Authorization':'Bearer '+session.access_token },
      body: JSON.stringify(stats),
    });
    if(!res.ok) throw new Error('submit failed: '+res.status);
    const { score, rank } = await res.json();
    setBanner('Score: '+score+'  (#'+rank+')'); setTimeout(()=>{ if(!gameWon) setBanner(''); },1800);
  } catch(e){
    setBanner("Couldn't save to leaderboard"); setTimeout(()=>{ if(!gameWon) setBanner(''); },1800);
  }
  refreshLeaderboard();
}
```
